% Font fallback (design 010 D4): a glyph the primary fonts lack (⊕ ⊙ ⌊ ⌋, sub/superscripts, box
% drawing, ✓ ✗ ...) comes from DejaVu Sans / DejaVu Sans Mono, everywhere the fonts are used (prose,
% code, headings, running heads, TOC, index). Glyphs the primary fonts have are unchanged.
% Quarto sets the main and mono fonts before this header; they are set again here, with the same
% default features, so that the fallback is attached. The last fallback= in a shape's features wins.
\directlua{
  luaotfload.add_fallback("pubfallback", {"DejaVu Sans:mode=node;", "DejaVu Sans Mono:mode=node;"})
  luaotfload.add_fallback("pubfallbackbold", {"DejaVu Sans/B:mode=node;", "DejaVu Sans Mono/B:mode=node;"})
  luaotfload.add_fallback("pubfallbackmono", {"DejaVu Sans Mono:mode=node;", "DejaVu Sans:mode=node;"})
  luaotfload.add_fallback("pubfallbackmonobold", {"DejaVu Sans Mono/B:mode=node;", "DejaVu Sans/B:mode=node;"})
}
\setmainfont{Atkinson Hyperlegible}[RawFeature={fallback=pubfallback},
  BoldFeatures={RawFeature={fallback=pubfallbackbold}},BoldItalicFeatures={RawFeature={fallback=pubfallbackbold}}]
\setmonofont{Fira Mono}[RawFeature={fallback=pubfallbackmono},
  BoldFeatures={RawFeature={fallback=pubfallbackmonobold}},BoldItalicFeatures={RawFeature={fallback=pubfallbackmonobold}}]
% Arrows, Greek, U+25B8 and U+2610 stay wrapped in \fallbackfont by panels.lua: this family is scaled

% A chapter's last short box (\pubfinalbox{content lines}, from panels.lua) must not sit alone on
% an otherwise empty page. When it would not fit, the page grows by less than four lines to take it;
% the box is then unbreakable and its own \needspace/\Needspace stand aside, because the growth
% already decided the page. (Estimate: content lines plus three for padding and skips.)
\newif\ifpubfinalbox
\newcommand{\pubfinalbox}[1]{\par\begingroup
  \dimen@=\numexpr#1+3\relax\baselineskip
  \ifdim\pagegoal<.5\maxdimen
    \dimen@ii\pagegoal\advance\dimen@ii-\pagetotal\advance\dimen@ii\pageshrink
    \advance\dimen@-\dimen@ii
    \ifdim\dimen@<4\baselineskip
      \global\pubfinalboxtrue
      \ifdim\dimen@>\z@\enlargethispage{\dimen@}\fi
    \fi
  \fi\endgroup}
